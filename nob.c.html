 67 static int ascii_tolower(int c) {
 68   return (c >= 'A' && c <= 'Z') ? (c - 'A' + 'a') : c;
 69 }
 70 
 71 static char *strstr_no_case(const char *h, const char *n) {
 72   if (!*n)
 73     return (char *)h;
 74   for (; *h; h++) {
 75     const char *h2 = h, *n2 = n;
 76     while (*h2 && *n2 && ascii_tolower(*h2) == ascii_tolower(*n2)) {
 77       h2++;
 78       n2++;
 79     }
 80     if (!*n2)
 81       return (char *)h;
 82   }
 83   return NULL;
 84 }

 90 enum bool_param { KEEP = -1, NO = 0, YES = 1 };
 91 void delete_entire_dir(const char *dirname);
 92 bool delete_one_file(Nob_Walk_Entry entry) {
 93   char *start = entry.data;
 94   switch (entry.type) {
 95   case FILE_DIRECTORY:
 96     if (strcmp(start, entry.path) != 0) {
 97       delete_entire_dir(entry.path);
 98       *entry.action = WALK_SKIP;
 99     }
100     break;
101   default:
102     delete_file(entry.path);
103     break;
104   }
105   return true;
106 }
107 void delete_entire_dir(const char *dirname) {
108   walk_dir(dirname, delete_one_file, (void *)dirname);
109   delete_file(dirname);

118 bool load_config(const char *fn, config *cnf) {
119   FILE *fp = fopen(fn, "rb");
120   *cnf = (config){0};
121   cnf->PTARGET = CURRENT_PLATFORM;
122   cnf->PFLOAT = F32;
123   if (!fp) {
124     nob_log(NOB_INFO, "couldn't load config, creating a default one: %s",
125             strerror(errno));
126     return false;
127   }
128   while (!feof(fp)) {
129     char line[4096];
130     char _[4096];
131 #define LOAD_CONFIG_CASE(val, param)                                           \
132   else if (sscanf(line, "#define " #param "_" #val "%s", _)) {                 \
133     (cnf)->P##param = val;                                                     \
134   }
135 #define LOAD_CONFIG(param, VALUES_VA) VALUES_VA(LOAD_CONFIG_CASE, param)
136 #define BOOL_PARAM_LOAD_CONFIG(param)                                          \
137   else if (sscanf(line, "#define PARAM_" #param "%s", _)) {                    \
138     (cnf)->PPARAM_##param = YES;                                               \
139   }
140 
141     if (fgets(line, sizeof(line), fp)) {
142       if (0) {
143       }
144       LOAD_CONFIG(TARGET, TARGET_VALUES_VA)
145       LOAD_CONFIG(FLOAT, FLOAT_VALUES_VA)
146       X_BOOL_PARAMS(BOOL_PARAM_LOAD_CONFIG)
147     } else {
148       break;
149     }
150   }
151   return true;
152 }
153 void merge_configs(config *c1, config *c2) {
154 #define MERGE_CONFIG(param, keep)                                              \
155   if ((c2)->P##param != keep)                                                  \
156     (c1)->P##param = (c2)->P##param;
157 #define BOOL_PARAM_MERGE_CONFIG(param) MERGE_CONFIG(PARAM_##param, KEEP)
158   MERGE_CONFIG(TARGET, TARGET_KEEP);
159   MERGE_CONFIG(FLOAT, FLOAT_KEEP);
160   X_BOOL_PARAMS(BOOL_PARAM_MERGE_CONFIG);
161 }
162 void print_config(config cnf) {
163 #define GET_DOC_SINGLE(val, doc, param, into)                                  \
164   if (cnf.P##param == val) {                                                   \
165     into = doc;                                                                \
166   }
167 #define GET_DOC(param, into, VALUES_DOC_VA)                                    \
168   VALUES_DOC_VA(GET_DOC_SINGLE, param, into)
169 #define PRINT_CASE(val, doc, ...)                                              \
170   case val:                                                                    \
171     s = #val;                                                                  \
172     break;
173 #define NONBOOL_PARAM_PRINT(param, VALUES_DOC_VA)                              \
174   do {                                                                         \
175     char *s, *doc;                                                             \
176     switch (cnf.P##param) {                                                    \
177       VALUES_DOC_VA(PRINT_CASE, 0);                                            \
178     default:                                                                   \
179       s = "KEEP";                                                              \
180       break;                                                                   \
181     }                                                                          \
182     GET_DOC(param, doc, VALUES_DOC_VA)                                         \
183     nob_log(NOB_INFO, #param "=%s (%s)", s, doc);                              \
184   } while (0);
185 #define BOOL_PARAM_PRINT(param, doc)                                           \
186   do {                                                                         \
187     char *s;                                                                   \
188     switch (cnf.PPARAM_##param) {                                              \
189       BOOL_VALUES_VA(PRINT_CASE, 0, 0);                                        \
190     default:                                                                   \
191       s = "KEEP";                                                              \
192       break;                                                                   \
193     }                                                                          \
194     if (cnf.PPARAM_##param == YES) {                                           \
195       nob_log(NOB_INFO, #param "=%s (%s)", s, doc);                            \
196     } else {                                                                   \
197       nob_log(NOB_INFO, #param "=%s", s);                                      \
198     }                                                                          \
199   } while (0);
200 
201   nob_log(NOB_INFO, "======= Config =======");
202   NONBOOL_PARAM_PRINT(TARGET, TARGET_VALUES_DOC_VA);
203   NONBOOL_PARAM_PRINT(FLOAT, FLOAT_VALUES_DOC_VA);
204   X_BOOL_PARAMS_DOC(BOOL_PARAM_PRINT);
205   nob_log(NOB_INFO, "======================");
206 }

219 bool dump_config(char *fn, config *cnf) {
220   FILE *fp = fopen(fn, "wb");
221   if (!fp) {
222     nob_log(NOB_ERROR, "Couldn't dump config file: %s", strerror(errno));
223     return false;
224   }
225 #define DUMP_CONFIG_CASE(val, doc, param)                                      \
226   if ((cnf)->P##param != val) {                                                \
227     fprintf(fp, "// ");                                                        \
228   }                                                                            \
229   fprintf(fp, "#define " #param "_" #val " //" doc "\n");
230 #define NONBOOL_PARAM_DUMP_CONFIG(param, VALUES_DOC_VA)                        \
231   VALUES_DOC_VA(DUMP_CONFIG_CASE, param)
232 #define BOOL_PARAM_DUMP_CONFIG(param, doc)                                     \
233   if ((cnf)->PPARAM_##param != YES) {                                          \
234     fprintf(fp, "// ");                                                        \
235   }                                                                            \
236   fprintf(fp, "#define PARAM_" #param " //" doc "\n");
237   fprintf(fp, "/* ----- available targets ----- */\n");
238   NONBOOL_PARAM_DUMP_CONFIG(TARGET, TARGET_VALUES_DOC_VA);
239   fprintf(fp, "/* -----   float sizes     ----- */\n");
240   NONBOOL_PARAM_DUMP_CONFIG(FLOAT, FLOAT_VALUES_DOC_VA);
241   fprintf(fp, "/* ----- build parameters  ----- */\n");
242   X_BOOL_PARAMS_DOC(BOOL_PARAM_DUMP_CONFIG);
243   fclose(fp);
244   nob_log(NOB_INFO, "config saved successfully");
245   return true;
246 }
247 int main(int argc, char **argv) {
248   GO_REBUILD_URSELF(argc, argv);
249   Cmd cmd = {0};
250 #define BOOL_PARAM_DEFAULT(b) KEEP,
251   config cnf = {TARGET_KEEP, FLOAT_KEEP, X_BOOL_PARAMS(BOOL_PARAM_DEFAULT)};
252   config loaded_cnf;
253   if (!mkdir_if_not_exists("config")) {
254     return 1;
255   }
256   shift(argv, argc);
257   bool config_only = false;
258   while (argc > 0) {
259     char *arg = shift(argv, argc);
260     if (0) {
261     } // so that we can have else ifs
262 #define BOOL_PARAM_PARSE(b)                                                    \
263   else if (strstr_no_case((arg), #b "=") == arg) {                             \
264     char *v = (arg) + sizeof(#b);                                              \
265     if (strstr_no_case(v, "yes") == v) {                                       \
266       (cnf).PPARAM_##b = YES;                                                  \
267     } else if (strstr_no_case(v, "no") == v) {                                 \
268       (cnf).PPARAM_##b = NO;                                                   \
269     } else {                                                                   \
270       (cnf).PPARAM_##b = (bool)atoi(v); /*make it either NO or YES*/           \
271     }                                                                          \
272   }
273 #define PARSE_CASE(val, param)                                                 \
274   else if (strcmp_no_case(v, #val) == 0) {                                     \
275     (cnf).P##param = val;                                                      \
276   }
277 #define NONBOOL_PARAM_PARSE(param, VALUES_VA)                                  \
278   else if (strstr_no_case((arg), #param "=") == arg) {                         \
279     char *v = (arg) + sizeof(#param);                                          \
280     if (0) {                                                                   \
281     }                                                                          \
282     VALUES_VA(PARSE_CASE, param)                                               \
283     else {                                                                     \
284       nob_log(NOB_WARNING, "Invalid value for " #param ": %s. Ignoring", v);   \
285     }                                                                          \
286   }
287     if (strcmp_no_case(arg, "clean") == 0) {
288       return clean();
289     }
290     if (strcmp_no_case(arg, "config") == 0) {
291       config_only = true;
292     }
293     X_BOOL_PARAMS(BOOL_PARAM_PARSE)
294     NONBOOL_PARAM_PARSE(TARGET, TARGET_VALUES_VA)
295     NONBOOL_PARAM_PARSE(FLOAT, FLOAT_VALUES_VA)
296   }
297   load_config("./config/build_config.h", &loaded_cnf);
298   merge_configs(&loaded_cnf, &cnf);
299   print_config(loaded_cnf);
300   if (!dump_config("./config/build_config.h", &loaded_cnf)) {
301     return 1;
302   }
303   if (config_only) {
304     return 0;
305   }
306 
307   const char *stage2 = "./targets/stage2";
308   cmd_append(&cmd, NOB_REBUILD_URSELF(stage2, "./targets/stage2.c"));
309   if (!cmd_run(&cmd))
310     return 1;
311   cmd_append(&cmd, stage2);
312   if (!cmd_run(&cmd))
313     return 1;
314 }
